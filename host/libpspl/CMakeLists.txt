cmake_minimum_required(VERSION 3.16)
project(visiongrasp_pspl VERSION 1.0 LANGUAGES C)
find_package(Threads REQUIRED)
add_library(pspl SHARED src/pspl.c)
target_include_directories(pspl PUBLIC include)
target_compile_features(pspl PRIVATE c_std_11)
target_compile_options(pspl PRIVATE -Wall -Wextra -Wconversion -Werror)
target_link_libraries(pspl PRIVATE Threads::Threads)
install(TARGETS pspl LIBRARY DESTINATION lib)
install(FILES include/pspl.h include/registers.h DESTINATION include)
